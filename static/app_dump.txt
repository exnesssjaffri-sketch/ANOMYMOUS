 564:         if (!inp || !out) return;
 565:         const t = inp.value.trim();

 567:             out.innerHTML = '<span class="error-message">Error: Task cannot be empty</span>';

 571:         this.isSubmitting = true;
 572:         this.state.taskStatus = 'submitting';

 578:     async performSubmission(inp, out) {
 579:         try {
 580:             const r = await fetch('/task', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({task: inp.value}) });
 581:             if (r.status >= 400) {
 582:                 let e = 'HTTP ' + r.status;
 583:                 try { const d = await r.json(); e += ': ' + (d.error || d.message || ''); } catch { e += ': ' + r.statusText; }
 584:                 throw new Error(e);
 585:             }
 586:             const d = await r.json();
 587:             if (d.status !== 'accepted') throw new Error('Invalid: ' + d.status);
 588:             out.innerHTML = '<span class="error-message">Task accepted... Polling...</span>';
 589:             await this.pollStatus(out);
 590:         } catch(e) {
 591:             console.error('[Task] Error:', e);
 592:             this.isSubmitting = false;
 593:             this.state.taskStatus = 'failed';
 594:             out.className = 'status-output error';
 595:             out.innerHTML = '<span class="error-message">Error: ' + String(e) + '</span>';
 596:         }
 597:     }
 598: 
 599:     async pollStatus(out) {
 600:         this.pollingAttempts = 0;
 601:         const start = Date.now();
 602:         clearInterval(this.pollingInterval);
 603:         this.pollingInterval = setInterval(async () => {
 604:             this.pollingAttempts++;
 605:             if (Date.now() - start > 60000) {
 606:                 clearInterval(this.pollingInterval);
 607:                 this.isSubmitting = false;
 608:                 out.className = 'status-output error';
 609:                 out.innerHTML = '<span class="error-message">Task polling timed out</span>';
 610:                 return;
 611:             }
 612:             try {
 613:                 const r = await fetch('/status');
 614:                 if (!r.ok) throw new Error('HTTP ' + r.status);
 615:                 const d = await r.json();
 616:                 if (d.status === 'no_result') return;
 617:                 clearInterval(this.pollingInterval);
 618:                 this.isSubmitting = false;
 619:                 this.state.taskStatus = d.status;
 620:                 out.className = 'status-output ' + d.status;
 621:                 const outText = d.status === 'success' ? 'Success!' : 'Failed!';
 622:                 const outBody = d.output || d.result || d.execution || d.error || JSON.stringify(d);
 623:                 out.innerHTML = '<span class="error-message">' + outText + '</span><pre class="status-result">' + outBody + '</pre>';
 624:             } catch(e) {
 625:                 clearInterval(this.pollingInterval);
 626:                 this.isSubmitting = false;
 627:                 this.state.taskStatus = 'failed';
 628:                 out.className = 'status-output error';
 629:                 out.innerHTML = '<span class="error-message">Polling error: ' + String(e) + '</span>';
 630:             }
 631:         }, 1000);
 632:     }
